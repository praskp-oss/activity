Lihat Kode Google Apps Script (Code.gs) 
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Laporan");
    if (!sheet) sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];

    var data = JSON.parse(e.postData.contents);
    var folderName = "Bukti_Laporan_Sales";
    var folders = DriveApp.getFoldersByName(folderName);
    var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);

    function saveImage(base64Data, taskName) {
      if (!base64Data) return "-";
      var parts = base64Data.split(",");
      var contentType = parts[0].split(":")[1].split(";")[0];
      var bytes = Utilities.base64Decode(parts[1]);
      var fileName = data.tanggal + "_" + data.namaCRO.replace(/\s+/g, "_") + "_" + taskName + ".jpg";
      var blob = Utilities.newBlob(bytes, contentType, fileName);
      var file = folder.createFile(blob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      return file.getUrl();
    }

    var linkFU = saveImage(data.buktiFollowUp, "FollowUp_ApplyLimit");
    var linkPosting = saveImage(data.buktiPosting, "Posting");
    var linkLive = saveImage(data.buktiLive, "LiveTikTok");

    sheet.appendRow([
      new Date(),
      data.tanggal,
      data.namaCRO,
      data.statusFollowUp ? "Selesai" : "Belum",
      data.jumlahFollowUp || 0,
      linkFU,
      data.statusPosting ? "Selesai" : "Belum",
      (data.platformPosting || "-") + " (" + (data.jumlahPosting || 0) + " konten)",
      linkPosting,
      data.statusLive ? "Selesai" : "Belum",
      data.durasiLive || 0,
      linkLive
    ]);

    return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
